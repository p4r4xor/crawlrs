deploys flip to S3-compatible storage on the prod chart directly.

# Wait for everything to come up
kubectl wait --for=condition=ready pod \
  -l app.kubernetes.io/instance={{ .Release.Name }} \
  -n {{ .Release.Namespace }} --timeout=180s

# Tail crawler logs
kubectl logs -f -n {{ .Release.Namespace }} {{ .Release.Name }}-0

# List the running services (use these names for port-forwarding)
kubectl get svc -n {{ .Release.Namespace }}

# Port-forward Grafana
kubectl port-forward -n {{ .Release.Namespace }} \
  svc/{{ .Release.Name }}-grafana 3000:3000
# Then http://localhost:3000  (admin / admin); dashboards live in the
# "crawlrs" folder.

# Port-forward the crawler's metrics + healthz endpoints
kubectl port-forward -n {{ .Release.Namespace }} \
  sts/{{ .Release.Name }} 9090:9090
# Then curl http://localhost:9090/metrics
#       curl http://localhost:9090/readyz

# Inspect blobs the crawler has written so far
kubectl exec -n {{ .Release.Namespace }} {{ .Release.Name }}-0 -- \
  ls -la /var/lib/crawlrs/data

#######################################################################
# Sandbox-shape disclaimers                                           #
#                                                                     #
# - persistence: Redis + Postgres have small PVCs. Crawled blobs land #
#   in an emptyDir by default and are LOST on pod restart; flip       #
#   crawlrs.store.backend.persistence.enabled=true for a PVC instead. #
# - credentials: hard-coded ("crawlrs" / "crawlrs"). DO NOT expose    #
#   this chart to the internet.                                       #
# - replicas: single replica everywhere. No HA, no failover.          #
# - resource limits: tuned for laptop / kind clusters.                #
#                                                                     #
# For production use the bare `charts/crawlrs/` chart and bring       #
# external Redis / Postgres + flip store.backend.kind=s3 with your    #
# own S3 bucket + creds.                                              #
#######################################################################
